```

```

## 四、复跑观测的命令（速查）

```bash
# 手机：抓屏 + 控件树 + 可读控件表
source ops/观测/_tools/env.sh        # 内含 ADB/PHONE/OBS 变量与 cap/tap/sw/back/txt 函数
cap "70-某屏"                         # 生成 .png/.xml/.table.txt 三件套

# 桌面：带调试端口重启客户端（之后可用 cdp.mjs 直读 DOM）
powershell -ExecutionPolicy Bypass -File ops/观测/_tools/launch-debug.ps1
node ops/观测/_tools/cdp.mjs eval "({title:document.title})"
powershell -ExecutionPolicy Bypass -File ops/观测/_tools/shot-window.ps1 -Out <绝对路径.png>
# 观测结束后恢复普通启动（关掉调试端口）
powershell -ExecutionPolicy Bypass -File ops/观测/_tools/launch-normal.ps1

# 网页：Tabbit CLI（Playwright）
"$LOCALAPPDATA/Tabbit/LocalAgent/bin/tabbit-cli.exe" create --task "FlowUs 观测"
```
